git init — создаёт новый локальный репозиторий
git add — добавляет файлы в индекс (Staging Area)
git commit — сохраняет изменения в историю
git status — показывает состояние рабочей директории
git log — показывает историю коммитов
git push — отправляет изменения на удаленный сервер
git pull — скачивает изменения с удаленного сервера
